// ============================================================================
// TEMPLATE - Quick Idea Capture
//
// LLM RULE:
// DON'T USE <> TAGS in comments. It breaks further rendering of markdown in obsidian.
//
// WHAT THIS DOES:
// Routes an idea note to either:
//   - Project Root ideas/
//   - Phase-specific phases/NNN/ideas/
//
// TWO MODES:
// - MANUAL: Uses the interactive workflow below.
// - AUTO: Called by \[T]phase.md for a note already at its final path; skips
//   prompts, moving, and title editing while rendering the same content.
//
// HOW IT WORKS:
// 1. Discovers the project note (\_project.md).
// 2. Builds a list of destination folders (project root + each phase's ideas/).
// 3. Prompts user to pick the destination folder.
// 4. Creates target folder if needed.
// 5. Finds a unique name (untitled\_idea, untitled\_idea\_2, etc.).
// 6. Sets dynamic backlink: \[← Plan] if in a phase, or \[← Project] if at root.
// 7. Moves note and opens title editor for immediate naming.
// ============================================================================

// ============================================================================
// 0. CONFIG — the single place to adjust names / paths / prompts.
//    NOTE: Other templates have their own CFG block; if a SHARED value changes
//    (projectFileName, phasesDirName, ideasDirName, backlinks, ...),
//    update it there as well.
//    autoCaptureContext must match \[T]phase.md and \[T]reflections.md.
// ============================================================================
const CFG = {
// --- Project discovery ----------------------------------------------------
projectFileName: "\_project.md",         // marker note that defines a project root
promptSelectProject: "Select Project:",

```
// --- Folder names -----------------------------------------------------------
phasesDirName: "phases",
ideasDirName: "ideas",

// --- Suggester labels & prompts ----------------------------------------------
rootLabel: "[Project Root]",
phaseLabelPrefix: "[Phase ",
promptSelectDestination: "Select ideas destination:",

// --- Default note naming (collision-safe: "..._2", "..._3", ...) --------------
defaultNoteName: "untitled_idea",
counterSeparator: "_",

// --- Navigation links (relative to the note's final location) -----------------
backlinkToPhasePlan: "[← Plan](../_plan.md)",
backlinkToProject: "[← Project](../_project.md)",

// --- Programmatic generation ------------------------------------------------
autoCaptureContext: "__phaseCaptureAuto", // applies only to the specified child file

// --- UX / environment -----------------------------------------------------------
titleEditCommand: "workspace:edit-file-title",
titleEditDelayMs: 200,
ctimeFormat: "YYYY-MM-DD HH:mm",
```

};

// Exact-path matching prevents automatic mode from affecting an unrelated note.
const autoContext = window\[CFG.autoCaptureContext];
const isAutoGenerated = autoContext?.targetPath === tp.config.target\_file.path;
let selectedDestination;
let finalName;

if (isAutoGenerated) {
// create\_new has already placed this note at its final destination.
const targetFolder = tp.config.target\_file.parent.path;
const folderSegments = targetFolder.split("/");
selectedDestination = {
path: targetFolder,
isPhase: folderSegments.length >= 3
&& folderSegments\[folderSegments.length - 3] === CFG.phasesDirName
};
} else {
// 1. Discover all project files in the vault
const projectFiles = app.vault.getMarkdownFiles()
.filter(f => f.name === CFG.projectFileName);

```
if (projectFiles.length === 0) {
    new Notice(`No '${CFG.projectFileName}' found in vault!`);
    return;
}

// Select Project (auto-select if only 1 project exists)
let selectedProjectFile = projectFiles[0];
if (projectFiles.length > 1) {
    selectedProjectFile = await tp.system.suggester(
        projectFiles.map(f => f.parent ? f.parent.path : f.path),
        projectFiles,
        false,
        CFG.promptSelectProject
    );
    if (!selectedProjectFile) return;
}

const projectRoot = selectedProjectFile.parent ? selectedProjectFile.parent.path : "";
const phasesDirPath = projectRoot ? `${projectRoot}/${CFG.phasesDirName}` : CFG.phasesDirName;
const rootIdeasPath = projectRoot ? `${projectRoot}/${CFG.ideasDirName}` : CFG.ideasDirName;

// 2. Build destination list: Root ideas + each Phase's ideas
const destinations = [];

destinations.push({
    label: `${CFG.rootLabel} ${CFG.ideasDirName}`,
    path: rootIdeasPath,
    isPhase: false
});

const allFolders = app.vault.getAllLoadedFiles().filter(f => f.children);
const phaseFolders = allFolders
    .filter(f => f.path.startsWith(phasesDirPath) && f.path !== phasesDirPath && f.parent && f.parent.path === phasesDirPath)
    .sort((a, b) => b.name.localeCompare(a.name, undefined, { numeric: true }));

phaseFolders.forEach(phase => {
    destinations.push({
        label: `${CFG.phaseLabelPrefix}${phase.name}] ${CFG.ideasDirName}`,
        path: `${phase.path}/${CFG.ideasDirName}`,
        isPhase: true
    });
});

// 3. Prompt user to select destination folder
selectedDestination = await tp.system.suggester(
    destinations.map(d => d.label),
    destinations,
    false,
    CFG.promptSelectDestination
);

if (!selectedDestination) return;

// 4. Ensure target folder exists on disk
if (!app.vault.getAbstractFileByPath(selectedDestination.path)) {
    await app.vault.createFolder(selectedDestination.path);
}

// 5. Collision-safe placeholder name (e.g. untitled_idea, untitled_idea_2)
let baseName = CFG.defaultNoteName;
finalName = baseName;
let counter = 1;

while (app.vault.getAbstractFileByPath(`${selectedDestination.path}/${finalName}.md`)) {
    counter++;
    finalName = `${baseName}${CFG.counterSeparator}${counter}`;
}
```

}

// 6. Compute dynamic navigation backlink
const backlink = selectedDestination.isPhase ? CFG.backlinkToPhasePlan : CFG.backlinkToProject;

if (!isAutoGenerated) {
// 7. Move note and open title editor for instant renaming
await tp.file.move(`${selectedDestination.path}/${finalName}`);

```
setTimeout(() => {
    app.commands.executeCommandById(CFG.titleEditCommand);
}, CFG.titleEditDelayMs);
```

}
