// DON'T USE <> TAGS in comments. It breaks further rendering of markdown in obsidian.

// WHAT THIS DOES:
// Creates or populates the "\_plan\_NNN.md" file for a phase, where NNN is the
// phase number (e.g. phases/004/\_plan\_004.md) so plan titles are unique vault-wide.
//
// TWO MODES:
// 1. AUTO: Called by \[T]phase.md. The note is already in the right folder.
//    It skips all prompts and immediately renders the markdown content below.
// 2. MANUAL: You opened a blank note and applied this template yourself.
//    It prompts you to pick the Project and Phase, checks for existing plans,
//    moves the note to phases/NNN/\_plan\_NNN.md, and renders the content.
// ================================================================

// ================================================================
// 0. CONFIG — the single place to adjust names / paths / prompts.
//    SHARED VALUES (keep in sync with the other templates' CFG blocks):
//      - CFG.autoPlanFlag : must equal CFG.autoPlanFlag in \[T]phase.md
//                           (they share the flag through `window`).
// ================================================================
const CFG = {
// --- Phase plan note
planFileNamePrefix: "\_plan", // note is named `${planFileNamePrefix}_${phaseNumber}` (no extension)

```
// --- Auto-mode flag (set by [T]phase.md before including this template)
autoPlanFlag: "__phasePlanAuto",

// --- Project & phase discovery
projectFileName: "_project.md",
promptSelectProject: "Select Project:",
phasesDirName: "phases",

// --- Navigation (final note location: project/phases/NNN/_plan.md)
backlinkToProject: "[← Project](../../_project.md)",

// --- Dataview render helper
reflectionHelpersPath: "_static/scripts/reflection_helpers.js",

// --- UX / environment
ctimeFormat: "YYYY-MM-DD HH:mm",
```

};

// If the note is already named as a plan (\_plan\_NNN) OR the auto-mode flag is true, skip prompts.
const isAutoGenerated = tp.file.title.startsWith(CFG.planFileNamePrefix) || window\[CFG.autoPlanFlag] === true;

if (!isAutoGenerated) {
// 1. Find all project files (\_project.md) in the vault
const projectFiles = app.vault.getMarkdownFiles()
.filter(f => f.name === CFG.projectFileName);

```
if (projectFiles.length === 0) {
    new Notice(`No '${CFG.projectFileName}' found in vault!`);
    return;
}

// Select Project (auto-select if only 1 project exists)
let selectedProjectFile = projectFiles[0];
if (projectFiles.length > 1) {
    selectedProjectFile = await tp.system.suggester(
        projectFiles.map(f => f.parent ? f.parent.path : f.path),
        projectFiles,
        false,
        CFG.promptSelectProject
    );
    if (!selectedProjectFile) return; // User pressed Escape
}

const projectRoot = selectedProjectFile.parent ? selectedProjectFile.parent.path : "";
const phasesDirPath = projectRoot ? `${projectRoot}/${CFG.phasesDirName}` : CFG.phasesDirName;

// 2. Find all phase folders under this project (sorted newest first, e.g. 003, 002, 001)
const allFolders = app.vault.getAllLoadedFiles().filter(f => f.children);
const phaseFolders = allFolders
    .filter(f => f.path.startsWith(phasesDirPath) && f.path !== phasesDirPath && f.parent && f.parent.path === phasesDirPath)
    .sort((a, b) => b.name.localeCompare(a.name, undefined, { numeric: true }));

if (phaseFolders.length === 0) {
    new Notice(`No phases found under ${phasesDirPath}!`);
    return;
}

// Prompt user to pick which phase needs a plan note
const selectedPhase = await tp.system.suggester(
    phaseFolders.map(f => f.name),
    phaseFolders,
    false,
    `Select Phase for this plan (${CFG.planFileNamePrefix}_NNN):`
);
if (!selectedPhase) return;

// 3. Collision check: Stop if a plan already exists and has written content
//    Plan note name embeds the phase number so plan titles stay unique vault-wide.
const planNoteName = `${CFG.planFileNamePrefix}_${selectedPhase.name}`;
const targetPath = `${selectedPhase.path}/${planNoteName}.md`;
const existingPlan = app.vault.getAbstractFileByPath(targetPath);
if (existingPlan && existingPlan.stat.size > 0) {
    new Notice(`${planNoteName}.md already exists in ${selectedPhase.name}! Aborted.`);
    return;
}
// If an existing file is 0 bytes (empty placeholder), move it to trash before replacing
if (existingPlan) {
    await app.vault.trash(existingPlan, true);
}

// 4. Move this note into the chosen phase folder as "_plan_NNN.md"
await tp.file.move(`${selectedPhase.path}/${planNoteName}`);
```

}
